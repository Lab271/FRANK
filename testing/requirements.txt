## Installation Commands

# Basic installation
pip install pytest requests boto3 cryptography

# Full testing suite
pip install -r testing/requirements.txt

# Development environment
pip install -r testing/requirements-dev.txt

## Environment Setup

### AWS Configuration
export AWS_ACCESS_KEY_ID=your_access_key
export AWS_SECRET_ACCESS_KEY=your_secret_key
export AWS_DEFAULT_REGION=us-west-2

### FRANK Platform Configuration
export FRANK_API_URL=https://api.frank.example.com
export FRANK_AUTH_TOKEN=your_auth_token
export FRANK_TEST_ENV=staging

### Database Configuration (if applicable)
export FRANK_DB_URL=postgresql://user:pass@localhost/frank_test
export FRANK_REDIS_URL=redis://localhost:6379/0

### Testing Configuration
export PYTEST_WORKERS=4  # Parallel test execution
export TEST_TIMEOUT=300   # Test timeout in seconds
export COVERAGE_THRESHOLD=80  # Minimum coverage percentage

## Docker Environment (Alternative)

# Use Docker for isolated testing environment
FROM python:3.11-slim

# Install system dependencies
RUN apt-get update && apt-get install -y \
    curl \
    netcat \
    postgresql-client \
    && rm -rf /var/lib/apt/lists/*

# Install Python dependencies
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy test files
COPY . /testing
WORKDIR /testing

# Run tests
CMD ["pytest", "--html=reports/test_report.html", "--cov=.", "--cov-report=html"]
